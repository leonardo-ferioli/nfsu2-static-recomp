cmake_minimum_required(VERSION 3.20)
project(nfsu2_raw_static LANGUAGES CXX)
add_link_options(-fuse-ld=lld)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/nue_d3d9/include")
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "This raw lowering uses Clang's 128-bit integers and x87 inline assembly")
endif()
file(GLOB RAW_UNITS CONFIGURE_DEPENDS generated/page_*.cpp)
if(NOT RAW_UNITS)
    message(FATAL_ERROR "Run tools/generate.py first")
endif()
# Audio layer (docs/NUE_HANDOVER_DSOUND.md): DirectSound semantics (Level A), spatial renderers, sinks, guest facade. Portable C++; the only
# Windows file is the WASAPI sink. Builds in both the 32-bit and the 64-bit configuration.
set(AUDIO_SOURCES platform/audio/ds_device.cpp platform/audio/spatial_renderers.cpp platform/audio/reverb.cpp platform/audio/null_sink.cpp
    platform/audio/audio_engine.cpp platform/audio/dsound_facade.cpp)
if(WIN32)
    list(APPEND AUDIO_SOURCES platform/audio/win32/wasapi_sink.cpp)
endif()
add_library(nfsu2_audio STATIC ${AUDIO_SOURCES})
target_include_directories(nfsu2_audio PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(nfsu2_audio PRIVATE -O2 -g0 -fno-fast-math -ffp-contract=off)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    # SSE arithmetic like the 64-bit build (no x87 extended precision): the rendered output hashes must agree between the two.
    target_compile_options(nfsu2_audio PRIVATE -msse2 -mfpmath=sse)
endif()
if(WIN32)
    target_link_libraries(nfsu2_audio PUBLIC ole32 avrt)
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    # Ghidra's SLEIGH engine (Apache 2.0), used only to interpret code that is generated at run time (dynamic_code.hpp).
    file(GLOB SLEIGH_SOURCES third_party/sleigh/*.cc)
    add_library(sleigh_runtime STATIC ${SLEIGH_SOURCES})
    target_include_directories(sleigh_runtime PRIVATE third_party/zlib_shim third_party/sleigh)
    target_compile_definitions(sleigh_runtime PRIVATE _WINDOWS)
    target_compile_options(sleigh_runtime PRIVATE -O2 -w -Wno-c++11-narrowing)
    set_target_properties(sleigh_runtime PROPERTIES CXX_STANDARD 17)
    set_source_files_properties(dynamic_code.cpp PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sleigh;${CMAKE_CURRENT_SOURCE_DIR}/third_party/zlib_shim")
    set(PLATFORM_SOURCE native32.cpp native32_reservation.cpp speed.cpp dynamic_code.cpp platform/services.cpp platform/win32/win32_services.cpp platform/null/null_platform.cpp nue_d3d9/src/d3d9_trace.cpp nue_d3d9/src/null_gpu.cpp d3d9_recorder.cpp win32_facade.cpp win32_facade_dinput.cpp platform/audio/dsound_install.cpp)
    add_link_options(-Wl,/BASE:0x00400000 -Wl,/DYNAMICBASE:NO -Wl,/LARGEADDRESSAWARE:NO -Wl,/SAFESEH:NO "-Wl,/sectionlayout:@${CMAKE_CURRENT_SOURCE_DIR}/native32_sections.txt")
else()
    set(PLATFORM_SOURCE platform.cpp)
endif()
add_library(nfsu2_raw STATIC runtime.cpp userops.cpp ${PLATFORM_SOURCE} generated/dispatch.cpp ${RAW_UNITS})
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    target_link_libraries(nfsu2_raw PUBLIC legacy_win32_host)   # the Win32 facade (win32_facade.cpp) uses the Level A library and its Windows backends
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    target_compile_definitions(nfsu2_raw PUBLIC NFSU2_NATIVE32=1)
endif()
target_include_directories(nfsu2_raw PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set(RAW_OPT_LEVEL "2" CACHE STRING "Optimization level for the statically translated runtime")
target_compile_options(nfsu2_raw PRIVATE "-O${RAW_OPT_LEVEL}" -g0 -fno-fast-math -fno-strict-aliasing)
target_link_libraries(nfsu2_raw PUBLIC bcrypt user32 winmm nfsu2_audio)
if(TARGET sleigh_runtime)
    target_link_libraries(nfsu2_raw PUBLIC sleigh_runtime)
endif()
# Differential-test instrumentation lives in runtime.cpp for every backend; fast pages opt in per library.
set_source_files_properties(runtime.cpp PROPERTIES COMPILE_DEFINITIONS NFSU2_DIFF_TRACE=1)
# Fast backend (generated_fast/): one function per page, local register file, native integers.
# 32-bit only: it relies on guest addresses being identical host addresses.
set(FAST_DIR "generated_fast" CACHE STRING "Directory holding the generated fast pages (use generated_fast_dev for subset experiments)")
if(CMAKE_SIZEOF_VOID_P EQUAL 4 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${FAST_DIR}/dispatch.cpp")
    file(GLOB FAST_UNITS CONFIGURE_DEPENDS ${FAST_DIR}/page_*.cpp)
    foreach(variant fast fast_trace)
        set(exclude "")
        if(variant STREQUAL "fast_trace")
            set(exclude EXCLUDE_FROM_ALL)  # differential builds only: cmake --build --target fast_resource_trace_tests
        endif()
        add_library(nfsu2_${variant} STATIC ${exclude} runtime.cpp userops.cpp ${PLATFORM_SOURCE} ${FAST_DIR}/dispatch.cpp ${FAST_DIR}/mods_tables.cpp mods_runtime.cpp mods_core.cpp mods_resolution.cpp mods_rendering.cpp mods_frontend.cpp mods_anchor.cpp mods_system.cpp mods_input.cpp mods_nos.cpp mods_registry.cpp mods_gamma.cpp mods_widescreen.cpp mods_detail.cpp ${FAST_UNITS})
        target_compile_definitions(nfsu2_${variant} PUBLIC NFSU2_NATIVE32=1 NFSU2_FAST_BACKEND=1)
        target_link_libraries(nfsu2_${variant} PUBLIC legacy_win32_host)
        target_include_directories(nfsu2_${variant} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_options(nfsu2_${variant} PRIVATE "-O${RAW_OPT_LEVEL}" -g0 -fno-fast-math -fno-strict-aliasing -mfpmath=sse -Wno-unused-label -ffunction-sections)
        target_link_libraries(nfsu2_${variant} PUBLIC bcrypt user32 winmm shell32 advapi32 nfsu2_audio)
        target_link_libraries(nfsu2_${variant} PUBLIC sleigh_runtime)
    endforeach()
    target_compile_definitions(nfsu2_fast_trace PRIVATE NFSU2_DIFF_TRACE=1)
    add_executable(speed_fast runner.cpp)
    target_link_libraries(speed_fast PRIVATE nfsu2_fast psapi)
    target_link_options(speed_fast PRIVATE "-Wl,/MAP:$<TARGET_FILE_DIR:speed_fast>/speed_fast.map")
    # Product launcher: reads the user's SPEED2.EXE at run time (no game bytes inside the program).
    add_executable(dynamic_decode_dump EXCLUDE_FROM_ALL dynamic_decode_dump.cpp)
    target_link_libraries(dynamic_decode_dump PRIVATE nfsu2_fast)
    # The guest image blocks as data inside the launcher: it does not need SPEED2.EXE at run time.
    find_package(Python3 COMPONENTS Interpreter)
    set(IMAGE_DATA_CPP "${CMAKE_BINARY_DIR}/image_data.cpp")
    file(GLOB IMAGE_BLOCK_FILES "${CMAKE_CURRENT_SOURCE_DIR}/export/memory_*.bin")
    add_custom_command(OUTPUT "${IMAGE_DATA_CPP}"
        COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_image.py" "${CMAKE_CURRENT_SOURCE_DIR}/generated/platform_image.inc" "${CMAKE_CURRENT_SOURCE_DIR}/export" "${IMAGE_DATA_CPP}"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_image.py" "${CMAKE_CURRENT_SOURCE_DIR}/generated/platform_image.inc" ${IMAGE_BLOCK_FILES} VERBATIM)
    add_executable(nfsu2_launcher WIN32 launcher.cpp "${IMAGE_DATA_CPP}")
    target_include_directories(nfsu2_launcher PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    set_source_files_properties("${IMAGE_DATA_CPP}" PROPERTIES COMPILE_OPTIONS "-Wno-c23-extensions;-Wno-c++-extensions")
    target_link_libraries(nfsu2_launcher PRIVATE nfsu2_fast)
    # Symbol map of the launcher: the performance report names code by address, this resolves it to the translated page it belongs to.
    target_link_options(nfsu2_launcher PRIVATE "-Wl,/MAP:$<TARGET_FILE_DIR:nfsu2_launcher>/nfsu2_launcher.map")
    # Speed layer: the translated code that race profiles found hot is laid out together (tools/hot_order.py writes the list; every
    # function is in its own section). Same code, fewer instruction-cache and TLB misses. -DSPEED_ORDER_FILE= turns it off.
    set(SPEED_ORDER_FILE "${CMAKE_CURRENT_SOURCE_DIR}/tools/hot_order.txt" CACHE FILEPATH "Linker order file of the hot translated code (empty: none)")
    if(SPEED_ORDER_FILE AND EXISTS "${SPEED_ORDER_FILE}")
        target_link_options(nfsu2_launcher PRIVATE "-Wl,/order:@${SPEED_ORDER_FILE}" "-Wl,/ignore:4037")
        set_property(TARGET nfsu2_launcher APPEND PROPERTY LINK_DEPENDS "${SPEED_ORDER_FILE}")
    endif()
    # Declared dependency: the SLEIGH specification the runtime P-code interpreter reads (dynamic_code.hpp).
    add_custom_command(TARGET nfsu2_launcher POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sleigh_spec/x86.sla" "$<TARGET_FILE_DIR:nfsu2_launcher>/sleigh/x86.sla")
    add_executable(speed_fast_trace EXCLUDE_FROM_ALL runner.cpp)
    target_compile_definitions(speed_fast_trace PRIVATE NFSU2_DIFF_TRACE=1)
    target_link_libraries(speed_fast_trace PRIVATE nfsu2_fast_trace)
    add_executable(fast_resource_tests resource_tests.cpp)
    target_link_libraries(fast_resource_tests PRIVATE nfsu2_fast)
    add_executable(fast_resource_trace_tests EXCLUDE_FROM_ALL resource_tests.cpp)
    target_compile_definitions(fast_resource_trace_tests PRIVATE NFSU2_DIFF_TRACE=1)
    target_link_libraries(fast_resource_trace_tests PRIVATE nfsu2_fast_trace)
    add_executable(fast_platform_tests native32_tests.cpp)
    target_link_libraries(fast_platform_tests PRIVATE nfsu2_fast)
    add_executable(d3d_shadow_tests d3d_shadow_tests.cpp)
    # Platform API (docs/PLATFORM_API.md): interfaces against the null and Win32 implementations, and the method-token facade registry.
    add_executable(platform_api_tests platform/api_tests.cpp platform/services.cpp platform/win32/win32_services.cpp platform/null/null_platform.cpp nue_d3d9/src/d3d9_trace.cpp nue_d3d9/src/null_gpu.cpp)
    target_include_directories(platform_api_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    add_executable(fast_facade_tests platform/facade_tests.cpp)
    add_executable(fast_win32_facade_tests platform/win32_facade_tests.cpp)
    target_link_libraries(fast_win32_facade_tests PRIVATE nfsu2_fast)
    add_executable(fast_dsound_install_tests platform/audio/dsound_install_test.cpp)
    target_link_libraries(fast_dsound_install_tests PRIVATE nfsu2_fast)
    add_executable(fast_recorder_tests platform/recorder_tests.cpp)
    target_link_libraries(fast_recorder_tests PRIVATE nfsu2_fast d3d9)
    target_link_libraries(fast_facade_tests PRIVATE nfsu2_fast)
    add_executable(fast_widescreen_tests widescreen_tests.cpp)
    target_link_libraries(fast_widescreen_tests PRIVATE nfsu2_fast)
    add_executable(fast_seh_tests native32_seh_tests.cpp)
    target_link_libraries(fast_seh_tests PRIVATE nfsu2_fast)
    add_executable(fast_x87_diff x87_diff_main.cpp)
    target_link_libraries(fast_x87_diff PRIVATE nfsu2_fast)
    add_executable(fast_native_bench EXCLUDE_FROM_ALL native_call_bench.cpp)
    target_link_libraries(fast_native_bench PRIVATE nfsu2_fast psapi)
    target_link_options(fast_native_bench PRIVATE "-Wl,/MAP:$<TARGET_FILE_DIR:fast_native_bench>/fast_native_bench.map")
    add_executable(fast_dynamic_tests dynamic_code_tests.cpp)
    target_link_libraries(fast_dynamic_tests PRIVATE nfsu2_fast)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/generated_optests/optests_table.cpp")
        # Operation-level differential test (python tools/gen_fast_optests.py first).
        add_executable(fast_optests EXCLUDE_FROM_ALL optests_main.cpp generated_optests/optests_ref.cpp
            generated_optests/optests_fast.cpp generated_optests/optests_table.cpp)
        target_compile_options(fast_optests PRIVATE -O2 -mfpmath=sse -Wno-unused-label)
        target_link_libraries(fast_optests PRIVATE nfsu2_fast)
    endif()
endif()
add_executable(speed_raw runner.cpp)
